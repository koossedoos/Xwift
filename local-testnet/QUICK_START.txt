  XWIFT LOCAL TESTNET - QUICK START
═══════════════════════════════════════════════════════════════════

Prerequisites:
✓ Ubuntu 20.04+ or compatible Linux
✓ Docker & Docker Compose installed
✓ 16GB+ RAM, 400GB+ storage, 6+ CPU cores

═══════════════════════════════════════════════════════════════════
STEP 1: Install Docker (if not already installed)
═══════════════════════════════════════════════════════════════════

curl -fsSL https://get.docker.com -o get-docker.sh
sudo sh get-docker.sh
sudo usermod -aG docker $USER
# Log out and back in

═══════════════════════════════════════════════════════════════════
STEP 2: Navigate to testnet directory
═══════════════════════════════════════════════════════════════════

cd local-testnet

═══════════════════════════════════════════════════════════════════
STEP 3: Create wallet
═══════════════════════════════════════════════════════════════════

# Build Docker image first
docker build -f ../Dockerfile.testnet -t local-xwift:testnet ..

# Create wallet
mkdir -p wallet
docker run --rm -it -v $(pwd)/wallet:/wallet local-xwift:testnet \
  xwift-wallet-cli --testnet --generate-new-wallet /wallet/miner-wallet

# SAVE YOUR 25-WORD SEED PHRASE!
# Copy your wallet address (starts with "X...")

═══════════════════════════════════════════════════════════════════
STEP 4: Configure environment
═══════════════════════════════════════════════════════════════════

cp .env.example .env
nano .env
# Set: MINER_ADDRESS=<your_wallet_address_from_step_3>

═══════════════════════════════════════════════════════════════════
STEP 5: Start testnet
═══════════════════════════════════════════════════════════════════

./start-testnet.sh

# Wait 60 seconds for initialization

═══════════════════════════════════════════════════════════════════
STEP 6: Monitor
═══════════════════════════════════════════════════════════════════

# Real-time dashboard
./monitoring/monitor-nodes.sh

# Quick status
./status-testnet.sh

# View logs
./logs-testnet.sh mining --follow
